在软件工程中，设计模式被分为多种类型，以解决不同的设计问题。其中，根据作用范围，设计模式可以被分为**类模式**和**对象模式**。它们之间的主要区别在于它们如何使用继承和组合来解决设计问题。

### 类模式
+ **定义**：类模式处理的是类与其子类之间的关系。它们通过继承来实现，这意味着它们在编译时刻就已经确定了。类模式主要通过多态性来实现代码的复用。
+ **关键点**：类模式关注的是类的结构和它们之间的静态关系。它们使用继承机制来组织代码和算法。
+ **例子**：工厂方法（Factory Method）设计模式就是一个典型的类模式。在这个模式中，一个类通过定义一个创建对象的接口，让其子类决定实例化哪一个类。工厂方法使得一个类的实例化延迟到其子类。

### 对象模式
+ **定义**：对象模式处理的是对象之间的关系。它们通过组合来实现，这意味着它们可以在运行时刻动态地改变。对象模式主要关注的是做事的方式（即算法和责任分配），更多地利用对象组合而不是继承。
+ **关键点**：对象模式关注的是对象之间的动态关系，即对象如何相互协作完成任务。这些模式通常通过对象的属性引用其他对象来实现。
+ **例子**：观察者（Observer）设计模式是一个典型的对象模式。在这个模式中，一个对象（称为subject）维护一系列依赖于它的对象（称为observers），当subject的状态变化时，会自动通知所有的observers。

### 类模式与对象模式的区别
1. **继承与组合**：类模式主要通过继承实现，强调的是类之间的静态关系。而对象模式主要通过组合实现，强调的是对象间的动态关系。
2. **编译时刻与运行时刻**：类模式的关系在编译时确定，而对象模式的关系可以在运行时动态变化。
3. **静态与动态**：类模式更静态，关注点在于固定的结构。对象模式更动态，关注点在于对象间的交互和协作。
4. **复用方式**：类模式通过继承实现代码复用，而对象模式通过对象的组合和委托实现复用。

```java
import java.util.ArrayList;

/**
 * 生成器模式
 */

public class BuilderMain {

        Director director = new Director();
        Builder builder = new ConcreteBuilder();

class Director {

        builder.buildPartB();

        product.add("Part A");
    }

    @Override
    public void buildPartB() {
        product.add("Part B");
    }

    @Override
    public void buildPartC() {
        product.add("Part C");
    }

    @Override
    public Product getResult() {
        return product;
    }
}

class Product {
    ArrayList<String> parts = new ArrayList<>();

    public void add(String part) {
        parts.add(part);
    }

    public List<String> getParts() {
        return parts;
```

### 代码实现
```java
package DesignPattern.Prototype;

public class PrototypeMain {
    public static void main(String[] args) {
        ConcretePrototype prototype = new ConcretePrototype(1, 100.0, "Prototype");
        ConcretePrototype prototype1 = (ConcretePrototype) prototype.Clone();
        System.out.println(prototype1.getId());
        System.out.println(prototype1.getPrice());
        System.out.println(prototype1.getName());
    }
}

interface Prototype {
    public Object Clone();
}

class ConcretePrototype implements Prototype {
    private int id;
    private double price;
    private String name;

    public ConcretePrototype() {};
    public ConcretePrototype(int id, double price, String name) {
        this.id = id;
        this.price = price;
        this.name = name;
    }

    public int getId() {
        return id;
    }

    public double getPrice() {
        return price;
    }

    public String getName() {
        return name;
    }

    @Override
    public Object Clone() {
        ConcretePrototype prototype = new ConcretePrototype();
        prototype.id = this.id;
        prototype.price = this.price;
        prototype.name = this.name;
        return prototype;
    }
}
```

```java
public class CompositeMain {

abstract class Component {

    public void PrintName() {
        System.out.println(name);

    public abstract boolean Remove(Component component);
}

    public Leaf(String name) {
        this.name = name;
    }

    @Override
    public boolean Add(Component component) {
       return false;
    }

    @Override
    public boolean Remove(Component component) {
        return false;
    }
}

class Composite extends Component {
    private List<Component> childrenList = new ArrayList<Component>();

    public Composite(String name) {
        this.name = name;
    }


    @Override
    public boolean Add(Component component) {
        return childrenList.add(component);
    }

    @Override
    public boolean Remove(Component component) {
       return childrenList.remove(component);
    }
}
```

```java
        flyweight2.operation("外在状态2");
        // 这个操作将具有与 flyweight1 相同的内在状态
        flyweight3.operation("外在状态3");

        // 使用非共享的具体享元对象
        Flyweight unsharedFlyweight = new UnsharedConcreteFlyweight("全部状态");
        unsharedFlyweight.operation("外在状态4");
    }
}

// 享元接口
interface Flyweight {
    void operation(String extrinsicState);
}

// 具体享元类
class ConcreteFlyweight implements Flyweight {
    private String intrinsicState; // 内在状态

    public ConcreteFlyweight(String intrinsicState) {
        this.intrinsicState = intrinsicState;
    }

    @Override
    public void operation(String extrinsicState) {
        System.out.println("内在状态 = " + intrinsicState + ", 外在状态 = " + extrinsicState);
    }
}

// 非共享的具体享元类
class UnsharedConcreteFlyweight implements Flyweight {
    private String allState; // 全部状态

    public UnsharedConcreteFlyweight(String allState) {
        this.allState = allState;
    }

    @Override
    public void operation(String extrinsicState) {
        System.out.println("全部状态 = " + allState + ", 外在状态 = " + extrinsicState);
    }
}

// 享元工厂类
class FlyweightFactory {
    private Map<String, Flyweight> flyweights = new HashMap<>(); // 维护享元对象的缓存

    public Flyweight getFlyweight(String key) {
        if (flyweights.containsKey(key)) {
            return flyweights.get(key); // 如果存在，则返回已有的享元对象
        } else {
            Flyweight newFlyweight = new ConcreteFlyweight(key); // 创建新的享元对象
            flyweights.put(key, newFlyweight); // 添加到缓存中
            return newFlyweight; // 返回新创建的享元对象
        }
    }
}
```

```java
package DesignPattern.Structural.Interpreter;

import java.util.HashMap;
import java.util.Map;

/**
 * 解释器模式
 */
public class InterpreterMain {
    public static void main(String[] args) {
        Context context = new Context();
        // 设置变量和值
        context.setVariable("a", 10);
        context.setVariable("b", 20);

        // 构建抽象语法树，(a + b)
        AbstractExpression expression = new NonterminalExpression(
                new TerminalExpression("a"),
                new TerminalExpression("b")
        );

        // 解释表达式
        int result = expression.interpret(context);

        System.out.println("(a + b) = " + result);
    }
}

// 上下文类，包含解释器之外的全局信息
class Context {
    private Map<String, Integer> variables = new HashMap<>();

    public void setVariable(String name, int value) {
        variables.put(name, value);
    }

    public int getVariable(String name) {
        return variables.getOrDefault(name, 0);
    }
}

// 抽象表达式类
abstract class AbstractExpression {
    public abstract int interpret(Context context);
}

// 终结符表达式类
class TerminalExpression extends AbstractExpression {
    private String variable;

    public TerminalExpression(String variable) {
        this.variable = variable;
    }

    @Override
    public int interpret(Context context) {
        return context.getVariable(variable);
    }
}

// 非终结符表达式类
class NonterminalExpression extends AbstractExpression {
    private AbstractExpression leftExpression;
    private AbstractExpression rightExpression;

    public NonterminalExpression(AbstractExpression left, AbstractExpression right) {
        this.leftExpression = left;
        this.rightExpression = right;
    }

    @Override
    public int interpret(Context context) {
        // 这里是一个简单的加法运算的例子
        return leftExpression.interpret(context) + rightExpression.interpret(context);
    }
}
```

```java
package DesignPattern.Structural.Iterator;


// 客户端类
public class IteratorDemo {
    public static void main(String[] args) {
        Object[] items = {"Item 1", "Item 2", "Item 3"};
        ConcreteAggregate aggregate = new ConcreteAggregate(items);
        Iterator iterator = aggregate.createIterator();

        for (iterator.first(); !iterator.isDone(); iterator.next()) {
            Object item = iterator.currentItem();
            System.out.println(item);
        }
    }
}

// 迭代器接口
interface Iterator {
    void first();
    void next();
    boolean isDone();
    Object currentItem();
}

// 聚合接口
interface Aggregate {
    Iterator createIterator();
}

// 具体聚合类
class ConcreteAggregate implements Aggregate {
    private Object[] items;

    public ConcreteAggregate(Object[] items) {
        this.items = items;
    }

    @Override
    public Iterator createIterator() {
        return new ConcreteIterator(this);
    }

    public int size() {
        return items.length;
    }

    public Object get(int index) {
        if (index < items.length) {
            return items[index];
        }
        return null;
    }
}

// 具体迭代器类
class ConcreteIterator implements Iterator {
    private ConcreteAggregate aggregate;
    private int currentIndex = 0;

    public ConcreteIterator(ConcreteAggregate aggregate) {
        this.aggregate = aggregate;
        first();
    }

    @Override
    public void first() {
        currentIndex = 0;
    }

    @Override
    public void next() {
        if (currentIndex < aggregate.size()) {
            currentIndex++;
        }
    }

    @Override
    public boolean isDone() {
        return currentIndex >= aggregate.size();
    }

    @Override
    public Object currentItem() {
        if (isDone()) {
            throw new IllegalStateException("Iterator out of bounds");
        }
        return aggregate.get(currentIndex);
    }
}
```

### 代码实现
```java
package DesignPattern.Behavioral.TemplateMethod;

/**
 * 模版方法
 */
public class TemplateMethodDemo {
    public static void main(String[] args) {
        Person student = new Student();
        student.TemplateMethod();
        Person teacher = new Teacher();
        teacher.TemplateMethod();
    }
}


abstract class Person {
    public void TemplateMethod() {
        System.out.println("上课 去教室");
        PrimitiveOperation1();
        System.out.println("下课 回家");
        PrimitiveOperation2();
    }

    public abstract void PrimitiveOperation1();  // 原语操作1：上课过程 学生 -> 听课；老师 -> 讲课
    public abstract void PrimitiveOperation2(); // 原语操作2：作业 学生 -> 写作业，交作业； 老师-> 布置作业，批改作业
}

class Student extends Person {

    @Override
    public void PrimitiveOperation1() {
        System.out.println("上课过程：学生 -> 听课");
    }

    @Override
    public void PrimitiveOperation2() {
        System.out.println("作业：学生 -> 写作业，交作业");
    }
}

class Teacher extends Person {
    @Override
    public void PrimitiveOperation1() {
        System.out.println("上课过程：老师 -> 讲课");
    }

    @Override
    public void PrimitiveOperation2() {
        System.out.println("作业：老师 -> 布置作业，批改作业");
    }
}
```

### 代码实现
```java
package DesignPattern.Behavioral.Visitor;

import java.util.ArrayList;
import java.util.List;
/**
 * 访问者
 */
public class VisitorDemo {
    public static void main(String[] args) {
        ElementStruct elementStruct = new ElementStruct();
        ConcreteVisitorA visitorA = new ConcreteVisitorA();
        elementStruct.accept(visitorA);
        visitorA.showResult();

        ConcreteVisitorB visitorB = new ConcreteVisitorB();
        elementStruct.accept(visitorB);
        visitorB.showResult();
    }
}

interface Visitor {
    public void visit(Student student);
    public void visit(Teacher teacher);
}

class ConcreteVisitorA implements Visitor { // 访问者A 分别统计学生和老师的年龄总和
    private int studentAgeSum = 0;
    private int teacherAgeSum = 0;

    public void showResult() {
        System.out.println("学生年龄总和：" + studentAgeSum);
        System.out.println("老师年龄总和：" + teacherAgeSum);
    }

    @Override
    public void visit(Student student) {
        System.out.println("访问者A访问学生:" +student.getName() + " 年龄：" + student.getAge());
        studentAgeSum += student.getAge();
    }

    @Override
    public void visit(Teacher teacher) {
        System.out.println("访问者A访问老师："+ teacher.getName() + " 年龄：" + teacher.getAge());
        teacherAgeSum += teacher.getAge();
    }
}

class ConcreteVisitorB implements Visitor { // 访问者2 分别求出学生的最高成绩 以及 老师的最高工龄
    private int maxScore = 0;
    private int maxWorkYear = 0;

    public void showResult() {
        System.out.println("学生最高成绩：" + maxScore);
        System.out.println("老师最高工龄：" + maxWorkYear);
    }
    @Override
    public void visit(Student student) {
        System.out.println("访问者B访问学生："+ student.getName() + " 成绩："+ student.getScore());
        if (student.getScore() > maxScore) {
            maxScore = student.getScore();
        }
    }
    @Override
    public void visit(Teacher teacher) {
        System.out.println("访问者B访问老师："+ teacher.getName() + " 工龄：" + teacher.getWorkYear());
        if (teacher.getWorkYear() > maxWorkYear) {
            maxWorkYear = teacher.getWorkYear();
        }
    }
}

class ElementStruct {
    private List<Element> elements = new ArrayList<Element>();

    public ElementStruct() {
        elements.add(new Student("张三", 18, 100));
        elements.add(new Student("李四", 19, 90));
        elements.add(new Student("王五", 20, 80));

        elements.add(new Teacher("李老师", 30, 3));
        elements.add(new Teacher("黄老师", 35, 4));
        elements.add(new Teacher("赵老师", 40, 5));
    }

    public void accept(Visitor visitor) {
        for (Element element : elements) {
            element.accept(visitor);
        }
    }
}


abstract class Element {
    private String name;
    private int age;

    public Element (String name, int age) {
        this.name = name;
        this.age = age;
    }
    public abstract void accept(Visitor visitor);

    public String getName() {
        return name;
    }

    public int getAge() {
        return age;
    }
}

class Student extends Element {
    private int score;
    public Student(String name, int age, int score) {
        super(name, age);
        this.score = score;
    }

    public int getScore() {
        return score;
    }

    @Override
    public void accept(Visitor visitor) {
        visitor.visit(this);
    }

}

class Teacher extends Element {
    private int workYear;
    public Teacher(String name, int age, int workYear) {
        super(name, age);
        this.workYear = workYear;
    }

    public int getWorkYear() {
        return workYear;
    }

    public void accept(Visitor visitor) {
        visitor.visit(this);
    }
}
```
