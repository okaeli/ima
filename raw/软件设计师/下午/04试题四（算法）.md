# 分治
## N皇后问题
<img src="https://cdn.nlark.com/yuque/0/2024/png/8424314/1715173796181-5ab2d4bc-62d7-4d07-9e58-bcb5d2084905.png" width="1286" title="" crop="0,0,1,1" id="u916ed30f" class="ne-image">

```c
#include <stdio.h>
#include <math.h>
#include <stdbool.h>

#define N 4

int q[N + 1]; // 存储皇后的列号
int answer = 0; // 方案数

bool check(int j) {
    for (int i = 1; i < j; i++) {
        // 判断同一列 q[i] == q[j]
        // 判断同一条斜线 |i - j| == |q[i] - q[j]|
        if (q[i] == q[j] || abs(q[i] - q[j]) == abs(i-j)) {
            return false;
        }
    }

    return true;
}

void queen() {
    // 初始化
    for (int i = 1; i < N; i++) {
        q[i] = 0;
    }
    int j = 1; // 表示正在拜访第 j 个皇后
    while (j >= 1) {  // 防止回溯越界
        q[j] = q[j] + 1; // 让第j个皇后向后一列拜访

        while (!check(j) && q[j] <= N) {  // 不合法后向后移
            q[j] = q[j] + 1;
        }

        if (q[j] <= N) {  // 如果当前位置合法
            if (j == N) {
                // 获得一个解
                answer++;
                printf("循环方案%d: ", answer);

                for (int i=1;i<=N; i++) {
                    printf("%d ", q[i]);
                }
                printf("\n");

            } else {
                // 摆放下一个皇后
                j += 1;
            }
        } else {
            q[j] = 0; // 重置当前j行皇后的位置
            j -= 1; // 回溯
        }
    }
}

void recursiveQueen(int j) {
    for (int i=1; i<=N; i++) {
        q[j] = i;

        if (check(j)) {
            if (j == N) {
                answer += 1;
                printf("递归方案%d: ", answer);

                for (int k =1; k<=N; k ++) {
                    printf("%d ", q[k]);
                }
                printf("\n");
            } else {
                recursiveQueen(j + 1);
            }
        }
    }
}

int main() {
//    queen();
    recursiveQueen(1);
    return 0;
}
```



# 动态规划
## 0-1背包问题
<img src="https://cdn.nlark.com/yuque/0/2024/png/8424314/1715221661409-a7397a9b-0854-4ed3-a82c-bdb80ef8bdc4.png" width="1421.1111487577" title="" crop="0,0,1,1" id="u9b9821d6" class="ne-image">

```c
#include <stdio.h>

int max(int a, int b) {
    return (a > b) ? a : b;
}

// 动态规划函数
int knapsack(int value[], int weight[], int n, int capacity) {
    int dp[n+1][capacity+1]; // 创建动态规划表

    // 初始化动态规划表
    for (int i = 0; i <= n; i++) {
        for (int w = 0; w <= capacity; w++) {
            if (i == 0 || w == 0) {
                dp[i][w] = 0; // 边界条件
            } else if (weight[i-1] <= w) {
                dp[i][w] = max(value[i-1] + dp[i-1][w-weight[i-1]], dp[i-1][w]); // 状态转移方程
            } else {
                dp[i][w] = dp[i-1][w]; // 不选择当前物品
            }
        }
    }

    return dp[n][capacity]; // 返回最大价值
}

int main() {
    int value[] = {12, 10, 20, 15}; // 物品价值数组
    int weight[] = {2, 1, 3, 2}; // 物品重量数组
    int n = sizeof(value) / sizeof(value[0]); // 物品个数
    int capacity = 5; // 背包容量

    int max_value = knapsack(value, weight, n, capacity); // 调用动态规划函数

    printf("最大价值为：%d\n", max_value);

    return 0;
}
```
